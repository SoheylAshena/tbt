DATABASE SETUP FOR PREMADE VPN CONFIGS
======================================

1. Back up the PostgreSQL database.
2. Connect to the same database configured in the bot's .env file.
3. Run all SQL below once. It is safe to run again if the first run was interrupted.
4. Restart/build the bot after the SQL succeeds.

SQL
---

BEGIN;

CREATE TABLE IF NOT EXISTS products (
  id BIGSERIAL PRIMARY KEY,
  code TEXT NOT NULL UNIQUE,
  name TEXT NOT NULL,
  price BIGINT NOT NULL CHECK (price >= 0),
  active BOOLEAN NOT NULL DEFAULT TRUE,
  sort_order INTEGER NOT NULL DEFAULT 0,
  created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

INSERT INTO products (code, name, price, sort_order)
VALUES
  ('10gb', '10 گیگ یک ماهه', 20000, 10),
  ('unlimited', 'نامحدود یک ماهه', 99000, 20)
ON CONFLICT (code) DO UPDATE SET
  name = EXCLUDED.name,
  price = EXCLUDED.price,
  sort_order = EXCLUDED.sort_order,
  active = TRUE,
  updated_at = CURRENT_TIMESTAMP;

ALTER TABLE orders ADD COLUMN IF NOT EXISTS product_id BIGINT;
ALTER TABLE orders ADD COLUMN IF NOT EXISTS config_id BIGINT;

DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_constraint WHERE conname = 'orders_product_id_fkey') THEN
    ALTER TABLE orders
      ADD CONSTRAINT orders_product_id_fkey
      FOREIGN KEY (product_id) REFERENCES products(id);
  END IF;
END $$;

CREATE TABLE IF NOT EXISTS product_configs (
  id BIGSERIAL PRIMARY KEY,
  product_id BIGINT NOT NULL REFERENCES products(id),
  config_text TEXT NOT NULL UNIQUE,
  assigned_order_id BIGINT UNIQUE,
  assigned_user_id BIGINT,
  assigned_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_constraint WHERE conname = 'product_configs_assigned_order_id_fkey') THEN
    ALTER TABLE product_configs
      ADD CONSTRAINT product_configs_assigned_order_id_fkey
      FOREIGN KEY (assigned_order_id) REFERENCES orders(id);
  END IF;

  IF NOT EXISTS (SELECT 1 FROM pg_constraint WHERE conname = 'orders_config_id_fkey') THEN
    ALTER TABLE orders
      ADD CONSTRAINT orders_config_id_fkey
      FOREIGN KEY (config_id) REFERENCES product_configs(id);
  END IF;
END $$;

CREATE INDEX IF NOT EXISTS product_configs_available_idx
  ON product_configs (product_id, id)
  WHERE assigned_order_id IS NULL;

CREATE INDEX IF NOT EXISTS orders_user_pending_idx
  ON orders (user_id, created_at DESC)
  WHERE status = 'pending_payment';

COMMIT;


ADMIN COMMANDS
--------------

Only Telegram IDs listed in ADMIN_ID can use these commands.

Add a 10 GB config:
/addconfig 10gb vless://YOUR_CONFIG_HERE

Add an unlimited config:
/addconfig unlimited vless://YOUR_CONFIG_HERE

The config may also start on the next line, which is useful for long values:
/addconfig 10gb
vless://YOUR_CONFIG_HERE

Check available and sold inventory:
/stock


NOTES
-----

- Prices are stored in products.price and use toman, matching the existing bot.
- Change prices with SQL when needed, for example:
  UPDATE products SET price = 25000, updated_at = CURRENT_TIMESTAMP WHERE code = '10gb';
- A config is never deleted after sale. Its assigned_order_id, assigned_user_id, and
  assigned_at columns preserve the delivery audit trail.
- Payment and config assignment happen in one transaction. If no config is available,
  the user's balance is not changed.
